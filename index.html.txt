    <script>
        async function enviarPrediccion() {
            const age = document.getElementById('age').value;
            const tumor_size = document.getElementById('tumor_size').value;
            const survival_months = document.getElementById('survival_months').value;
            const resultDiv = document.getElementById('result');
            
            resultDiv.innerText = "Procesando...";
            
            try {
                const response = await fetch('http://127.0.0', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        Age: parseInt(age),
                        Tumor_Size: parseInt(tumor_size),
                        Survival_Months: parseInt(survival_months)
                    })
                });
                const data = await response.json();
                if(data.estado_supervivencia === "Alive") {
                    resultDiv.style.color = "green";
                    resultDiv.innerText = "Resultado: Paciente Viva (Alive)";
                } else {
                    resultDiv.style.color = "red";
                    resultDiv.innerText = "Resultado: Paciente Fallecida (Dead)";
                }
            } catch (error) {
                resultDiv.style.color = "orange";
                resultDiv.innerText = "Error: Asegúrate de tener encendida tu API en cmd";
            }
        }
    </script>
